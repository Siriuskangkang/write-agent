## 一、新增文件

### 1.1 编辑组件
**文件：`frontend/src/components/workbench/StyleTemplate/StyleTemplateEdit.tsx`**

```typescript
'use client';

import { useState } from 'react';
import { Card, Form, Input, InputNumber, Select, Button, Space, Divider } from 'antd';
import type { StyleFeatures } from './types';

interface StyleTemplateEditProps {
  initialFeatures: StyleFeatures;
  onSave: (features: StyleFeatures) => void;
  onCancel: () => void;
}

export function StyleTemplateEdit({ initialFeatures, onSave, onCancel }: StyleTemplateEditProps) {
  const [features, setFeatures] = useState<StyleFeatures>(initialFeatures);

  const updateStructure = (field: string, value: any) => {
    setFeatures({
      ...features,
      structure: { ...features.structure, [field]: value },
    });
  };

  const updateLanguage = (field: string, value: any) => {
    setFeatures({
      ...features,
      language: { ...features.language, [field]: value },
    });
  };

  const updateFormatting = (field: string, value: any) => {
    setFeatures({
      ...features,
      formatting: { ...features.formatting, [field]: value },
    });
  };

  const updateCitation = (field: string, value: any) => {
    setFeatures({
      ...features,
      citation: { ...features.citation, [field]: value },
    });
  };

  return (
    <Card title="编辑体例">
      <Form layout="vertical">
        <Divider>结构规范</Divider>

        <Form.Item label="章节格式">
          <Input
            value={features.structure?.chapterFormat}
            onChange={(e) => updateStructure('chapterFormat', e.target.value)}
            placeholder="如：第X章"
          />
        </Form.Item>

        <Form.Item label="小节格式">
          <Input
            value={features.structure?.sectionFormat}
            onChange={(e) => updateStructure('sectionFormat', e.target.value)}
            placeholder="如：X.X"
          />
        </Form.Item>

        <Form.Item label="最大层级">
          <InputNumber
            value={features.structure?.maxDepth}
            onChange={(value) => updateStructure('maxDepth', value)}
            min={1}
            max={5}
          />
        </Form.Item>

        <Divider>语言风格</Divider>

        <Form.Item label="语气">
          <Select
            value={features.language?.tone}
            onChange={(value) => updateLanguage('tone', value)}
          >
            <Select.Option value="formal">正式</Select.Option>
            <Select.Option value="informal">非正式</Select.Option>
          </Select>
        </Form.Item>

        <Form.Item label="人称">
          <Select
            value={features.language?.person}
            onChange={(value) => updateLanguage('person', value)}
          >
            <Select.Option value="first">第一人称</Select.Option>
            <Select.Option value="third">第三人称</Select.Option>
          </Select>
        </Form.Item>

        <Form.Item label="时态">
          <Select
            value={features.language?.tense}
            onChange={(value) => updateLanguage('tense', value)}
          >
            <Select.Option value="present">现在时</Select.Option>
            <Select.Option value="past">过去时</Select.Option>
          </Select>
        </Form.Item>

        <Divider>格式规范</Divider>

        <Form.Item label="段落缩进">
          <Select
            value={features.formatting?.paragraphIndent}
            onChange={(value) => updateFormatting('paragraphIndent', value)}
          >
            <Select.Option value={true}>是</Select.Option>
            <Select.Option value={false}>否</Select.Option>
          </Select>
        </Form.Item>

        <Form.Item label="行间距">
          <InputNumber
            value={features.formatting?.lineSpacing}
            onChange={(value) => updateFormatting('lineSpacing', value)}
            min={1}
            max={3}
            step={0.1}
          />
        </Form.Item>

        <Form.Item label="字体">
          <Input
            value={features.formatting?.fontFamily}
            onChange={(e) => updateFormatting('fontFamily', e.target.value)}
            placeholder="如：宋体"
          />
        </Form.Item>

        <Divider>引用规范</Divider>

        <Form.Item label="引用格式">
          <Input
            value={features.citation?.style}
            onChange={(e) => updateCitation('style', e.target.value)}
            placeholder="如：GB/T 7714-2015"
          />
        </Form.Item>

        <Form.Item label="正文引用格式">
          <Input
            value={features.citation?.inTextFormat}
            onChange={(e) => updateCitation('inTextFormat', e.target.value)}
            placeholder="如：[1]"
          />
        </Form.Item>

        <Space style={{ marginTop: 16 }}>
          <Button type="primary" onClick={() => onSave(features)}>
            保存
          </Button>
          <Button onClick={onCancel}>取消</Button>
        </Space>
      </Form>
    </Card>
  );
}
```

## 二、修改容器组件

### 2.1 修改 index.tsx
**文件：`frontend/src/components/workbench/StyleTemplate/index.tsx`**

**修改 ViewMode 类型：**
```typescript
type ViewMode = 'upload' | 'analyzing' | 'preview' | 'edit' | 'detail';
```

**新增状态：**
```typescript
const [editingFeatures, setEditingFeatures] = useState<StyleFeatures | null>(null);
```

**新增方法：**
```typescript
const handleEdit = () => {
  if (currentTemplate?.features) {
    setEditingFeatures(currentTemplate.features);
    setViewMode('edit');
  }
};

const handleSaveEdit = async (features: StyleFeatures) => {
  try {
    const { updateStyleTemplate } = await import('@/services/styleTemplateApi');
    await updateStyleTemplate(currentTemplate.id, { features });
    setCurrentTemplate({ ...currentTemplate, features });
    setViewMode('preview');
    message.success('体例已更新');
  } catch (error) {
    message.error('保存失败');
  }
};

const handleCancelEdit = () => {
  setEditingFeatures(null);
  setViewMode('preview');
};
```

**新增渲染：**
```typescript
{viewMode === 'edit' && editingFeatures && (
  <StyleTemplateEdit
    initialFeatures={editingFeatures}
    onSave={handleSaveEdit}
    onCancel={handleCancelEdit}
  />
)}
```
