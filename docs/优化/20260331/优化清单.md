#### 1.3 实现 Anthropic Provider

**文件：`backend/src/llm/providers/anthropic.provider.ts`**

```typescript
import Anthropic from '@anthropic-ai/sdk';
import { LLMProvider, LLMConfig } from '../llm.interface';

export class AnthropicProvider implements LLMProvider {
  private client: Anthropic;
  private model: string;

  constructor(config: LLMConfig) {
    this.client = new Anthropic({
      apiKey: config.apiKey,
      baseURL: config.baseURL,
    });
    this.model = config.model;
  }

  async *streamCompletion(prompt: string) {
    const stream = await this.client.messages.stream({
      model: this.model,
      max_tokens: 4096,
      messages: [{ role: 'user', content: prompt }],
    });

    for await (const event of stream) {
      yield event;
    }
  }

  async generateCompletion(prompt: string): Promise<string> {
    const response = await this.client.messages.create({
      model: this.model,
      max_tokens: 4096,
      messages: [{ role: 'user', content: prompt }],
    });

    return response.content[0].type === 'text'
      ? response.content[0].text
      : '';
  }
}
```

#### 1.4 实现 DeepSeek Provider

**文件：`backend/src/llm/providers/deepseek.provider.ts`**

```typescript
import OpenAI from 'openai';
import { LLMProvider, LLMConfig } from '../llm.interface';

export class DeepSeekProvider implements LLMProvider {
  private client: OpenAI;
  private model: string;

  constructor(config: LLMConfig) {
    this.client = new OpenAI({
      apiKey: config.apiKey,
      baseURL: config.baseURL,
    });
    this.model = config.model;
  }

  async *streamCompletion(prompt: string) {
    const stream = await this.client.chat.completions.create({
      model: this.model,
      messages: [{ role: 'user', content: prompt }],
      stream: true,
    });

    for await (const chunk of stream) {
      // 转换为统一格式
      if (chunk.choices[0]?.delta?.content) {
        yield {
          type: 'content_block_delta',
          delta: {
            type: 'text_delta',
            text: chunk.choices[0].delta.content,
          },
        };
      }
    }
  }

  async generateCompletion(prompt: string): Promise<string> {
    const response = await this.client.chat.completions.create({
      model: this.model,
      messages: [{ role: 'user', content: prompt }],
    });

    return response.choices[0]?.message?.content || '';
  }
}
```

#### 1.5 创建 LLM 工厂

**文件：`backend/src/llm/llm.factory.ts`**

```typescript
import { Injectable } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { LLMProvider } from './llm.interface';
import { AnthropicProvider } from './providers/anthropic.provider';
import { DeepSeekProvider } from './providers/deepseek.provider';

@Injectable()
export class LLMFactory {
  constructor(private configService: ConfigService) {}

  createProvider(): LLMProvider {
    const provider = this.configService.get('LLM_PROVIDER', 'anthropic');

    switch (provider) {
      case 'anthropic':
        return new AnthropicProvider({
          apiKey: this.configService.get('ANTHROPIC_API_KEY'),
          baseURL: this.configService.get('ANTHROPIC_BASE_URL'),
          model: this.configService.get('ANTHROPIC_MODEL', 'claude-sonnet-4-6'),
        });

      case 'deepseek':
        return new DeepSeekProvider({
          apiKey: this.configService.get('DEEPSEEK_API_KEY'),
          baseURL: this.configService.get('DEEPSEEK_BASE_URL'),
          model: this.configService.get('DEEPSEEK_MODEL', 'deepseek-chat'),
        });

      default:
        throw new Error(`Unsupported LLM provider: ${provider}`);
    }
  }
}
```

#### 1.6 修改 AgentService

**文件：`backend/src/agent/agent.service.ts`**

将现有的 Anthropic 客户端替换为 LLM 工厂：

```typescript
import { LLMFactory } from '../llm/llm.factory';
import { LLMProvider } from '../llm/llm.interface';

@Injectable()
export class AgentService {
  private llmProvider: LLMProvider;

  constructor(private llmFactory: LLMFactory) {
    this.llmProvider = this.llmFactory.createProvider();
  }

  async *streamCompletion(prompt: string) {
    yield* this.llmProvider.streamCompletion(prompt);
  }
}
```

## 2. 体例优化方案

### 2.1 需求描述

**当前问题：**
- 用户可以直接生成目录，不需要先上传体例
- 生成的内容没有遵循特定的体例规范

**优化目标：**
- 生成目录前必须先上传并分析体例
- 根据体例特征生成符合规范的文章结构
- 体例特征应用到目录、大纲、正文生成的全流程

private buildStyleContext(features: StyleFeatures): string {
  return `
## 体例规范

请严格遵循以下体例规范生成目录结构：

### 结构规范
- 章节编号格式：${features.structure?.chapterFormat || '第X章'}
- 小节编号格式：${features.structure?.sectionFormat || 'X.X'}
- 最大层级深度：${features.structure?.maxDepth || 3}

### 语言风格
- 语气：${features.language?.tone || 'formal'}
- 人称：${features.language?.person || 'third'}
- 时态：${features.language?.tense || 'present'}
  `.trim();
}
