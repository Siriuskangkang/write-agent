```typescript
import {
  Entity,
  PrimaryGeneratedColumn,
  Column,
  CreateDateColumn,
  UpdateDateColumn,
  ManyToOne,
  JoinColumn,
} from 'typeorm';
import { Project } from '../project/entities/project.entity';

export interface StyleFeatures {
  structure?: {
    chapterFormat?: string;
    sectionFormat?: string;
    maxDepth?: number;
  };
  language?: {
    tone?: string;
    person?: string;
    tense?: string;
  };
  formatting?: {
    paragraphIndent?: boolean;
    lineSpacing?: number;
    fontFamily?: string;
  };
  citation?: {
    style?: string;
    inTextFormat?: string;
    bibliographyFormat?: string;
  };
}

@Entity('style_templates')
export class StyleTemplate {
  @PrimaryGeneratedColumn('uuid')
  id: string;

  @Column({ type: 'varchar', length: 255 })
  name: string;

  @Column({ type: 'text', nullable: true })
  description: string;

  @Column({ type: 'uuid', name: 'project_id' })
  projectId: string;

  @ManyToOne(() => Project, { onDelete: 'CASCADE' })
  @JoinColumn({ name: 'project_id' })
  project: Project;

  @Column({ type: 'jsonb', name: 'reference_file_ids', default: [] })
  referenceFileIds: string[];

  @Column({ type: 'jsonb', nullable: true })
  features: StyleFeatures;

  @Column({
    type: 'enum',
    enum: ['pending', 'analyzing', 'completed', 'failed'],
    default: 'pending',
  })
  status: 'pending' | 'analyzing' | 'completed' | 'failed';

  @Column({ type: 'text', nullable: true, name: 'error_message' })
  errorMessage: string;

  @CreateDateColumn({ name: 'created_at' })
  createdAt: Date;

  @UpdateDateColumn({ name: 'updated_at' })
  updatedAt: Date;
}
```

## 5. Service 实现

### StyleTemplateService

```typescript
// src/style-template/style-template.service.ts
import { Injectable, NotFoundException, ConflictException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { InjectQueue } from '@nestjs/bull';
import { Queue } from 'bull';
import { StyleTemplate } from './entities/style-template.entity';
import { CreateStyleTemplateDto } from './dto/create-style-template.dto';
import { UpdateStyleTemplateDto } from './dto/update-style-template.dto';

@Injectable()
export class StyleTemplateService {
  constructor(
    @InjectRepository(StyleTemplate)
    private readonly styleTemplateRepository: Repository<StyleTemplate>,
    @InjectQueue('style-analyze')
    private readonly analyzeQueue: Queue,
  ) {}

  async create(dto: CreateStyleTemplateDto): Promise<StyleTemplate> {
    const template = this.styleTemplateRepository.create({
      name: dto.name,
      description: dto.description,
      projectId: dto.projectId,
      referenceFileIds: dto.referenceFileIds,
      status: 'pending',
    });

    return this.styleTemplateRepository.save(template);
  }

  async findAll(projectId: string): Promise<StyleTemplate[]> {
    return this.styleTemplateRepository.find({
      where: { projectId },
      order: { createdAt: 'DESC' },
    });
  }

  async findOne(id: string): Promise<StyleTemplate> {
    const template = await this.styleTemplateRepository.findOne({
      where: { id },
    });

    if (!template) {
      throw new NotFoundException(`StyleTemplate with ID ${id} not found`);
    }

    return template;
  }

  async update(id: string, dto: UpdateStyleTemplateDto): Promise<StyleTemplate> {
    const template = await this.findOne(id);

    Object.assign(template, dto);
    template.updatedAt = new Date();

    return this.styleTemplateRepository.save(template);
  }

  async remove(id: string): Promise<void> {
    const template = await this.findOne(id);
    await this.styleTemplateRepository.remove(template);
  }

  async triggerAnalysis(id: string): Promise<string> {
    const template = await this.findOne(id);

    if (template.status === 'analyzing') {
      throw new ConflictException('Template is already being analyzed');
    }

    template.status = 'analyzing';
    template.errorMessage = null;
    await this.styleTemplateRepository.save(template);

    const job = await this.analyzeQueue.add('analyze', {
      templateId: id,
    });

    return job.id.toString();
  }

  async updateAnalysisResult(
    id: string,
    features: any,
    status: 'completed' | 'failed',
    errorMessage?: string,
  ): Promise<void> {
    await this.styleTemplateRepository.update(id, {
      features,
      status,
      errorMessage,
      updatedAt: new Date(),
    });
  }
}
```

### 核心方法说明

| 方法 | 说明 |
|------|------|
| `create()` | 创建体例模板，初始状态为 `pending` |
| `findAll()` | 按项目 ID 查询模板列表，按创建时间倒序 |
| `findOne()` | 根据 ID 查询单个模板，不存在时抛出 404 |
| `update()` | 更新模板字段，自动更新 `updatedAt` |
| `remove()` | 删除模板 |
| `triggerAnalysis()` | 触发 AI 分析，将任务加入 Bull 队列，返回 jobId |
| `updateAnalysisResult()` | Worker 完成后更新分析结果 |

```typescript
// src/style-template/style-template.controller.ts
import {
  Controller,
  Get,
  Post,
  Patch,
  Delete,
  Body,
  Param,
  Query,
  UseGuards,
  Sse,
  MessageEvent,
} from '@nestjs/common';
import { JwtAuthGuard } from '../auth/guards/jwt-auth.guard';
import { StyleTemplateService } from './style-template.service';
import { StyleAnalyzer } from './style-analyzer';
import { CreateStyleTemplateDto } from './dto/create-style-template.dto';
import { UpdateStyleTemplateDto } from './dto/update-style-template.dto';
import { Observable, from, switchMap, map } from 'rxjs';

@Controller('style-templates')
@UseGuards(JwtAuthGuard)
export class StyleTemplateController {
  constructor(
    private readonly styleTemplateService: StyleTemplateService,
    private readonly styleAnalyzer: StyleAnalyzer,
  ) {}

  @Post()
  create(@Body() dto: CreateStyleTemplateDto) {
    return this.styleTemplateService.create(dto);
  }

  @Get()
  findAll(@Query('projectId') projectId: string) {
    return this.styleTemplateService.findAll(projectId);
  }

  @Get(':id')
  findOne(@Param('id') id: string) {
    return this.styleTemplateService.findOne(id);
  }

  @Patch(':id')
  update(@Param('id') id: string, @Body() dto: UpdateStyleTemplateDto) {
    return this.styleTemplateService.update(id, dto);
  }

  @Delete(':id')
  async remove(@Param('id') id: string) {
    await this.styleTemplateService.remove(id);
    return { message: 'Template deleted successfully' };
  }

  @Post(':id/analyze')
  @Sse()
  analyze(@Param('id') id: string): Observable<MessageEvent> {
    return from(this.styleTemplateService.findOne(id)).pipe(
      switchMap((template) =>
        this.styleAnalyzer.analyzeStream(template.id, template.referenceFileIds),
      ),
      map((event) => ({
        type: event.type,
        data: event.data,
      })),
    );
  }
}
```

2. `switchMap()` 切换到分析流

## 7. AI 分析器

### StyleAnalyzer

```typescript
// src/style-template/style-analyzer.ts
import { Injectable } from '@nestjs/common';
import { AgentService } from '../agent/agent.service';
import { FileService } from '../file/file.service';
import { Observable, from, mergeMap, map } from 'rxjs';

interface AnalyzeEvent {
  type: 'meta' | 'token' | 'done' | 'error';
  data: any;
}

@Injectable()
export class StyleAnalyzer {
  constructor(
    private readonly agentService: AgentService,
    private readonly fileService: FileService,
  ) {}

  analyzeStream(templateId: string, fileIds: string[]): Observable<AnalyzeEvent> {
    return from(this.prepareContext(fileIds)).pipe(
      mergeMap((context) => this.streamAnalysis(templateId, context)),
    );
  }

  private async prepareContext(fileIds: string[]): Promise<string> {
    const files = await Promise.all(
      fileIds.map((id) => this.fileService.findOne(id)),
    );

    const contents = await Promise.all(
      files.map(async (file) => {
        const chunks = await this.fileService.getFileChunks(file.id);
        return {
          filename: file.originalName,
          content: chunks.map((c) => c.content).join('\n\n'),
        };
      }),
    );

    return contents
      .map((f) => `## ${f.filename}\n\n${f.content}`)
      .join('\n\n---\n\n');
  }

  private streamAnalysis(
    templateId: string,
    context: string,
  ): Observable<AnalyzeEvent> {
    const prompt = this.buildPrompt(context);

    return new Observable<AnalyzeEvent>((subscriber) => {
      subscriber.next({
        type: 'meta',
        data: { templateId },
      });

      let buffer = '';

      this.agentService
        .generateStream(prompt, [])
        .subscribe({
          next: (chunk) => {
            if (chunk.type === 'content_block_delta') {
              const text = chunk.delta.text;
              buffer += text;
              subscriber.next({
                type: 'token',
                data: { content: text },
              });
            }
          },
          complete: () => {
            try {
              const features = this.parseFeatures(buffer);
              subscriber.next({
                type: 'done',
                data: { features },
              });
              subscriber.complete();
            } catch (error) {
              subscriber.error(error);
            }
          },
          error: (err) => {
            subscriber.next({
              type: 'error',
              data: { message: err.message },
            });
            subscriber.error(err);
          },
        });
    });
  }

  private buildPrompt(context: string): string {
    return `你是一位专业的教材编写体例分析专家。请分析以下参考文档，提取其写作体例特征。

## 参考文档

${context}

## 分析要求

请从以下维度分析体例特征：

1. **结构规范**
   - 章节编号格式（如"第X章"、"X.X"）
   - 层级深度（最多几级标题）
   - 标题命名规范

2. **语言风格**
   - 语气（正式/非正式）
   - 人称（第一/第三人称）
   - 时态（现在/过去时）

3. **格式规范**
   - 段落缩进
   - 行间距
   - 字体要求

4. **引用规范**
   - 引用格式（如 GB/T 7714-2015）
   - 正文引用格式（如 [1]）
   - 参考文献格式

## 输出格式

请以 JSON 格式输出，结构如下：

\`\`\`json
{
  "structure": {
    "chapterFormat": "第X章",
    "sectionFormat": "X.X",
    "maxDepth": 3
  },
  "language": {
    "tone": "formal",
    "person": "third",
    "tense": "present"
  },
  "formatting": {
    "paragraphIndent": true,
    "lineSpacing": 1.5,
    "fontFamily": "宋体"
  },
  "citation": {
    "style": "GB/T 7714-2015",
    "inTextFormat": "[1]",
    "bibliographyFormat": "numbered"
  }
}
\`\`\`

请直接输出 JSON，不要包含其他说明文字。`;
  }

  private parseFeatures(text: string): any {
    const jsonMatch = text.match(/```json\s*([\s\S]*?)\s*```/);
    if (jsonMatch) {
      return JSON.parse(jsonMatch[1]);
    }

    const cleaned = text.trim();
    if (cleaned.startsWith('{') && cleaned.endsWith('}')) {
      return JSON.parse(cleaned);
    }

    throw new Error('Failed to parse features from AI response');
  }
}
```

### Prompt 设计

Prompt 包含以下部分：

1. **角色定义**：专业的教材编写体例分析专家
2. **输入上下文**：参考文档内容
3. **分析维度**：结构、语言、格式、引用四个维度
4. **输出格式**：JSON Schema 示例
5. **约束条件**：直接输出 JSON，不要额外说明

### 结果解析

`parseFeatures()` 方法支持两种格式：

1. Markdown 代码块包裹的 JSON
2. 纯 JSON 文本

解析失败时抛出异常，由 Worker 捕获并更新状态为 `failed`。

## 8. Bull Worker

### AnalyzeWorker

```typescript
// src/style-template/workers/analyze.worker.ts
import { Processor, Process } from '@nestjs/bull';
import { Job } from 'bull';
import { Logger } from '@nestjs/common';
import { StyleTemplateService } from '../style-template.service';
import { StyleAnalyzer } from '../style-analyzer';

@Processor('style-analyze')
export class AnalyzeWorker {
  private readonly logger = new Logger(AnalyzeWorker.name);

  constructor(
    private readonly styleTemplateService: StyleTemplateService,
    private readonly styleAnalyzer: StyleAnalyzer,
  ) {}

  @Process('analyze')
  async handleAnalyze(job: Job): Promise<void> {
    const { templateId } = job.data;

    this.logger.log(`Starting analysis for template ${templateId}`);

    try {
      const template = await this.styleTemplateService.findOne(templateId);

      const features = await new Promise((resolve, reject) => {
        let result: any;

        this.styleAnalyzer
          .analyzeStream(templateId, template.referenceFileIds)
          .subscribe({
            next: (event) => {
              if (event.type === 'done') {
                result = event.data.features;
              }
            },
            complete: () => {
              if (result) {
                resolve(result);
              } else {
                reject(new Error('No features extracted'));
              }
            },
            error: (err) => reject(err),
          });
      });

      await this.styleTemplateService.updateAnalysisResult(
        templateId,
        features,
        'completed',
      );

      this.logger.log(`Analysis completed for template ${templateId}`);
    } catch (error) {
      this.logger.error(
        `Analysis failed for template ${templateId}: ${error.message}`,
        error.stack,
      );

      await this.styleTemplateService.updateAnalysisResult(
        templateId,
        null,
        'failed',
        error.message,
      );

      throw error;
    }
  }
}
```

### 队列处理流程

```
1. Controller 触发分析 → 更新状态为 analyzing
2. 任务加入 Bull 队列 style-analyze
3. Worker 消费任务
4. 调用 StyleAnalyzer.analyzeStream()
5. 订阅 Observable，等待 done 事件
6. 提取 features，更新状态为 completed
7. 如果失败，更新状态为 failed，记录错误信息
```

### 错误处理

- 捕获所有异常，记录日志
- 更新模板状态为 `failed`
- 保存错误信息到 `errorMessage` 字段
- 重新抛出异常，触发 Bull 重试机制（可配置）
