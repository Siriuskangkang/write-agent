## 1. 需求概述

### 1.1 功能目标

实现教材体例管理功能，允许用户上传体例文件，AI 自动分析体例特征，用户确认后应用到项目中，后续生成内容时自动遵循该体例规范。

### 1.2 核心功能点

| 功能模块 | 功能描述 | 优先级 |
|---------|---------|--------|
| 体例上传 | 支持上传 DOCX/PDF/TXT 格式的体例文件 | P0 |
| AI 分析 | 自动提取体例的结构、格式、语言风格、内容特征 | P0 |
| 结果预览 | 结构化展示分析结果，支持手动编辑 | P0 |
| 体例激活 | 将体例模板应用到项目，后续生成内容时自动遵循 | P0 |
| 体例管理 | 查看、编辑、删除、切换体例模板 | P1 |
| 模板库 | 预置常见出版社体例模板（后期优化） | P2 |

### 1.3 用户流程

```
用户进入项目工作台
  ↓
点击"体例管理"
  ↓
上传体例文件（DOCX/PDF/TXT）
  ↓
系统显示"AI 正在分析..."（异步队列处理）
  ↓
分析完成，展示结构化结果
  ↓
用户预览并确认（可手动编辑）
  ↓
点击"确认并应用"
  ↓
体例模板激活，提示"已应用"
  ↓
后续生成目录/大纲/正文时自动遵循体例规范
```

## 3. 数据库设计

### 3.1 新增表：`style_templates`

**表名**：`style_templates`
**说明**：存储体例模板信息及分析结果

| 字段名 | 类型 | 约束 | 默认值 | 说明 |
|--------|------|------|--------|------|
| id | VARCHAR(36) | PRIMARY KEY | UUID | 主键 |
| project_id | VARCHAR(36) | NOT NULL, FK | - | 所属项目 ID |
| name | VARCHAR(255) | NOT NULL | - | 体例名称 |
| source_file_id | VARCHAR(36) | FK, NULL | - | 关联的源文件 ID |
| features | JSON | NOT NULL | - | 体例特征（结构化 JSON） |
| prompt_template | TEXT | NOT NULL | - | 生成时注入的 Prompt 模板 |
| is_active | BOOLEAN | NOT NULL | false | 是否为当前激活的体例 |
| status | VARCHAR(20) | NOT NULL | 'draft' | 状态：draft/analyzing/ready/error |
| created_at | DATETIME | NOT NULL | CURRENT_TIMESTAMP | 创建时间 |
| updated_at | DATETIME | NOT NULL | CURRENT_TIMESTAMP | 更新时间 |

**索引**：
- `idx_style_templates_project_id` ON `project_id`
- `idx_style_templates_status` ON `status`
- `idx_style_templates_is_active` ON `project_id, is_active`

**外键**：
- `project_id` REFERENCES `projects(id)` ON DELETE CASCADE
- `source_file_id` REFERENCES `documents(id)` ON DELETE SET NULL

### 3.2 修改表：`projects`

**新增字段**：

| 字段名 | 类型 | 约束 | 默认值 | 说明 |
|--------|------|------|--------|------|
| active_style_template_id | VARCHAR(36) | FK, NULL | NULL | 当前激活的体例模板 ID |

**外键**：
- `active_style_template_id` REFERENCES `style_templates(id)` ON DELETE SET NULL

**索引**：
- `idx_projects_active_style_template_id` ON `active_style_template_id`

### 3.3 `features` 字段 JSON 结构

```typescript
interface StyleFeatures {
  structure: {
    chapterFormat: string;        // 章节格式，如："第{n}章"
    sectionFormat: string;         // 小节格式，如："{chapter}.{section}"
    hierarchy: string[];           // 层级，如：["章", "节", "小节"]
    paragraphOrganization: string; // 段落组织，如："总分总"
  };
  format: {
    titleStyle: string;            // 标题样式，如："加粗、居中、二号字"
    paragraphIndent: boolean;      // 段落缩进
    listFormat: string;            // 列表格式
    citationStyle: string;         // 引用格式，如："GB/T 7714"
  };
  language: {
    tone: string;                  // 语气，如："正式、客观"
    person: string;                // 人称，如："第三人称"
    termDensity: string;           // 术语密度，如："中等"
    sentenceStyle: string;         // 句式，如："长短句结合"
  };
  content: {
    exampleFrequency: string;      // 案例频率，如："每节2-3个案例"
    figureFormat: string;          // 图表格式
    exerciseFormat: string;        // 练习题格式
    highlightMethod: string;       // 知识点标注方式
  };
}
```


  ```
  name?: string (可选，默认为"体例分析-日期")
  ```


#### 4.4.3 Service 实现

**文件**：`backend/src/style-template/style-template.service.ts`

```typescript
import { Injectable, NotFoundException, BadRequestException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { InjectQueue } from '@nestjs/bull';
import { Queue } from 'bull';
import { StyleTemplate, StyleTemplateStatus } from './entities/style-template.entity.js';
import { FileService } from '../file/file.service.js';
import { UpdateStyleTemplateDto } from './dto/update-style-template.dto.js';

@Injectable()
export class StyleTemplateService {
  constructor(
    @InjectRepository(StyleTemplate)
    private readonly styleTemplateRepo: Repository<StyleTemplate>,
    @InjectQueue('style-analyze')
    private readonly analyzeQueue: Queue,
    private readonly fileService: FileService,
  ) {}

  async uploadAndAnalyze(
    file: Express.Multer.File,
    projectId: string,
    userId: string,
    name?: string,
  ): Promise<StyleTemplate> {
    // 1. 上传文件到 file 模块
    const uploadedFile = await this.fileService.uploadFile(file, projectId, userId);

    // 2. 创建 StyleTemplate 记录
    const template = this.styleTemplateRepo.create({
      project_id: projectId,
      source_file_id: uploadedFile.id,
      name: name || `体例分析-${new Date().toLocaleDateString()}`,
      status: StyleTemplateStatus.ANALYZING,
      features: {} as any, // 初始为空
      prompt_template: '',
      is_active: false,
    });

    await this.styleTemplateRepo.save(template);

    // 3. 加入 Bull 队列异步分析
    await this.analyzeQueue.add('analyze', {
      templateId: template.id,
      fileId: uploadedFile.id,
    });

    return template;
  }

  async findById(id: string): Promise<StyleTemplate> {
    const template = await this.styleTemplateRepo.findOne({ where: { id } });
    if (!template) {
      throw new NotFoundException('体例模板不存在');
    }
    return template;
  }

  async findByProject(projectId: string, status?: StyleTemplateStatus): Promise<StyleTemplate[]> {
    const where: any = { project_id: projectId };
    if (status) {
      where.status = status;
    }
    return this.styleTemplateRepo.find({ where, order: { created_at: 'DESC' } });
  }

  async update(id: string, dto: UpdateStyleTemplateDto): Promise<StyleTemplate> {
    const template = await this.findById(id);

    if (dto.name) {
      template.name = dto.name;
    }

    if (dto.features) {
      template.features = { ...template.features, ...dto.features };
    }

    return this.styleTemplateRepo.save(template);
  }

  async activate(id: string): Promise<StyleTemplate> {
    const template = await this.findById(id);

    if (template.status !== StyleTemplateStatus.READY) {
      throw new BadRequestException('只能激活已完成分析的体例模板');
    }

    // 将同项目的其他体例设为非激活
    await this.styleTemplateRepo.update(
      { project_id: template.project_id, is_active: true },
      { is_active: false },
    );

    // 激活当前体例
    template.is_active = true;
    await this.styleTemplateRepo.save(template);

    // 更新 project 表的 active_style_template_id
    await this.styleTemplateRepo.manager.query(
      `UPDATE projects SET active_style_template_id = ? WHERE id = ?`,
      [template.id, template.project_id],
    );

    return template;
  }

  async delete(id: string): Promise<void> {
    const template = await this.findById(id);

    if (template.is_active) {
      throw new BadRequestException('无法删除激活中的体例模板，请先激活其他体例');
    }

    await this.styleTemplateRepo.remove(template);
  }

  async updateStatus(
    id: string,
    status: StyleTemplateStatus,
    features?: any,
    promptTemplate?: string,
  ): Promise<void> {
    const update: any = { status };
    if (features) update.features = features;
    if (promptTemplate) update.prompt_template = promptTemplate;

    await this.styleTemplateRepo.update(id, update);
  }
}
```

#### 4.4.6 Bull Worker 实现

**文件**：`backend/src/style-template/analyze.worker.ts`

```typescript
import { Processor, Process } from '@nestjs/bull';
import { Job } from 'bull';
import { Injectable, Logger } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { StyleTemplate, StyleTemplateStatus } from './entities/style-template.entity.js';
import { StyleAnalyzer } from './analyzers/style-analyzer.js';
import { FileService } from '../file/file.service.js';

@Processor('style-analyze')
@Injectable()
export class AnalyzeWorker {
  private readonly logger = new Logger(AnalyzeWorker.name);

  constructor(
    @InjectRepository(StyleTemplate)
    private readonly styleTemplateRepo: Repository<StyleTemplate>,
    private readonly styleAnalyzer: StyleAnalyzer,
    private readonly fileService: FileService,
  ) {}

  @Process('analyze')
  async handleAnalyze(job: Job<{ templateId: string; fileId: string }>) {
    const { templateId, fileId } = job.data;

    try {
      this.logger.log(`开始分析体例模板: ${templateId}`);

      // 1. 获取文件内容
      const document = await this.fileService.getDocumentByFileId(fileId);
      if (!document || !document.content_text) {
        throw new Error('文件内容为空');
      }

      // 2. AI 分析体例特征
      const features = await this.styleAnalyzer.analyze(document.content_text);

      // 3. 生成 Prompt 模板
      const promptTemplate = this.styleAnalyzer.buildPromptTemplate(features);

      // 4. 更新数据库
      await this.styleTemplateRepo.update(templateId, {
        status: StyleTemplateStatus.READY,
        features,
        prompt_template: promptTemplate,
      });

      this.logger.log(`体例模板分析完成: ${templateId}`);
    } catch (error) {
      this.logger.error(`体例模板分析失败: ${templateId}`, error);

      await this.styleTemplateRepo.update(templateId, {
        status: StyleTemplateStatus.ERROR,
      });

      throw error;
    }
  }
}
```
